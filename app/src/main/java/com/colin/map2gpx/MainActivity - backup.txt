class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContent {
            Map2GpxTheme {
                val navController = rememberNavController()
                NavHost(navController = navController, startDestination = "main") {
                    composable("main") { GPXScreen(navController) }
                    composable("preview") { GPXPreviewScreen() }
                }
            }
        }

        // Request Bluetooth permission on Android 12+
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            if (ContextCompat.checkSelfPermission(
                    this,
                    Manifest.permission.BLUETOOTH_CONNECT
                ) != PackageManager.PERMISSION_GRANTED

                ActivityCompat.requestPermissions(
                    this,
                    arrayOf(Manifest.permission.BLUETOOTH_CONNECT),
                    100
                )
            }
        }
    }
}

@Composable
fun GPXScreen(navController: androidx.navigation.NavHostController) {
    val context = LocalContext.current
    val gpxContent = remember { mutableStateOf("") }
    val statusMessage = remember { mutableStateOf("") }
    val filename = remember { mutableStateOf("route.gpx") }

    Column(modifier = Modifier.padding(16.dp)) {
        Text("Map2GPX", style = MaterialTheme.typography.headlineMedium)
        Spacer(modifier = Modifier.height(16.dp))

        Button(onClick = {
            gpxContent.value = generateSampleGPX()
            statusMessage.value = "GPX generated."
        }) {
            Text("Generate GPX")
        }

        Spacer(modifier = Modifier.height(8.dp))

        TextField(
            value = filename.value,
            onValueChange = { filename.value = it },
            label = { Text("Filename") }
        )

        Spacer(modifier = Modifier.height(8.dp))

        Button(onClick = {
            val content = gpxContent.value
            if (content.isNotEmpty()) {
                val file = saveGPXToCache(context, content, filename.value)
                shareGPXFile(context, file)
                statusMessage.value = "GPX saved and shared."
            } else {
                statusMessage.value = "No GPX to save."
            }
        }) {
            Text("Save & Share GPX")
        }

        Spacer(modifier = Modifier.height(8.dp))

        Button(onClick = {
            val content = gpxContent.value
            if (content.isNotEmpty()) {
                saveGPXToDownloads(context, content, filename.value)
                statusMessage.value = "GPX saved to Downloads."
            } else {
                statusMessage.value = "No GPX to save."
            }
        }) {
            Text("Save to Downloads")
        }

        Spacer(modifier = Modifier.height(8.dp))

        Button(onClick = {
            val content = gpxContent.value
            if (content.isNotEmpty()) {
                // Pass GPX content via a singleton or shared state
                GPXPreviewState.gpxContent = content
                navController.navigate("preview")
            } else {
                statusMessage.value = "No GPX to preview."
            }
        }) {
            Text("Preview GPX Route")
        }

        Spacer(modifier = Modifier.height(16.dp))
        Text(statusMessage.value)
    }
}

fun generateSampleGPX(): String {
    return """
        <?xml version="1.0" encoding="UTF-8"?>
        <gpx version="1.1" creator="Map2GPX">
            <wpt lat="37.7749" lon="-122.4194"><name>San Francisco</name></wpt>
            <wpt lat="34.0522" lon="-118.2437"><name>Los Angeles</name></wpt>
        </gpx>
    """.trimIndent()
}

fun saveGPXToCache(context: Context, gpx: String, filename: String): File {
    val dir = File(context.cacheDir, "files")
    if (!dir.exists()) dir.mkdirs()
    val file = File(dir, filename)
    FileOutputStream(file).use { it.write(gpx.toByteArray()) }
    return file
}

fun saveGPXToDownloads(context: Context, gpx: String, filename: String): File {
    val dir = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
    if (!dir.exists()) dir.mkdirs()
    val file = File(dir, filename)
    FileOutputStream(file).use { it.write(gpx.toByteArray()) }
    return file
}

fun shareGPXFile(context: Context, file: File) {
    val authority = "${context.packageName}.fileprovider"
    val uri: Uri = FileProvider.getUriForFile(context, authority, file)

    val bluetoothIntent = Intent(Intent.ACTION_SEND).apply {
        type = "application/gpx+xml"
        putExtra(Intent.EXTRA_STREAM, uri)
        setPackage("com.android.bluetooth")
        addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
    }

    try {
        context.startActivity(Intent.createChooser(bluetoothIntent, "Send GPX via Bluetooth"))
    } catch (e: Exception) {
        val shareIntent = Intent(Intent.ACTION_SEND).apply {
            type = "application/gpx+xml"
            putExtra(Intent.EXTRA_STREAM, uri)
            addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
        }
        context.startActivity(Intent.createChooser(shareIntent, "Share GPX"))
    }
}

// Shared state for preview
object GPXPreviewState {
    var gpxContent: String = ""
}

@Composable
fun GPXPreviewScreen() {
    val gpxContent = GPXPreviewState.gpxContent
    val context = LocalContext.current
    val mapView = remember { MapView(context) }

    // Render the MapView inside Compose
    AndroidView(
        factory = { mapView },
        modifier = Modifier.fillMaxSize()
    )

    // Forward lifecycle events to MapView
    DisposableEffect(Unit) {
        mapView.onStart()
        onDispose {
            mapView.onStop()
            mapView.onDestroy()
        }
    }

    // Initialize map and style once
    mapView.getMapAsync { map ->
        map.setStyle("https://demotiles.maplibre.org/style.json") { style ->
            val gpx = parseGpxToGeoJson(gpxContent)
            val line = toLineString(gpx.trackPoints)

            // Sources
            val routeSourceId = "gpx-route"
            val waypointSourceId = "gpx-waypoints"

            if (line != null) {
                style.addSource(
                    GeoJsonSource(routeSourceId, org.maplibre.geojson.Feature.fromGeometry(line))
                )
            } else {
                style.addSource(
                    GeoJsonSource(routeSourceId, FeatureCollection.fromFeatures(arrayOf()))
                )
            }

            style.addSource(
                GeoJsonSource(
                    waypointSourceId,
                    FeatureCollection.fromFeatures(
                        gpx.waypoints.map { org.maplibre.geojson.Feature.fromGeometry(it) }
                    )
                )
            )

            // Optional: register marker image if style lacks "marker-15"
            // style.addImage(
            //     "marker-15",
            //     BitmapFactory.decodeResource(context.resources, R.drawable.map_marker)
            // )

            // Route line layer
            val routeLayer = org.maplibre.android.style.layers.LineLayer(
                routeSourceId + "-layer",
                routeSourceId
            ).withProperties(
                PropertyFactory.lineColor("#FF4D00"),
                PropertyFactory.lineWidth(4.0f),
                PropertyFactory.lineJoin("round"),
                PropertyFactory.lineCap("round")
            )
            style.addLayer(routeLayer)

            // Waypoint markers
            val waypointLayer = SymbolLayer(waypointSourceId + "-layer", waypointSourceId)
                .withProperties(
                    PropertyFactory.iconImage("marker-15"),
                    PropertyFactory.iconAllowOverlap(true),
                    PropertyFactory.iconSize(1.2f)
                )
            style.addLayer(waypointLayer)

            // Fit camera to all points
            val allPoints = when {
                gpx.trackPoints.isNotEmpty() -> gpx.trackPoints
                gpx.waypoints.isNotEmpty() -> gpx.waypoints
                else -> emptyList()
            }
            if (allPoints.isNotEmpty()) {
                val bounds = latLngBoundsFor(allPoints)
                map.animateCamera(
                    CameraUpdateFactory.newLatLngBounds(bounds, 60)
                )
            }
        }
    }
}

// --- GPX parsing utilities ---

data class GpxData(
    val waypoints: List<Point>,
    val trackPoints: List<Point>
)

fun parseGpxToGeoJson(gpx: String): GpxData {
    val factory = XmlPullParserFactory.newInstance()
    val parser = factory.newPullParser().apply {
        setInput(gpx.reader())
    }

    val waypoints = mutableListOf<Point>()
    val trackPoints = mutableListOf<Point>()

    var eventType = parser.eventType
    while (eventType != XmlPullParser.END_DOCUMENT) {
        if (eventType == XmlPullParser.START_TAG) {
            when (parser.name.lowercase()) {
                "wpt" -> {
                    val lat = parser.getAttributeValue(null, "lat")?.toDoubleOrNull()
                    val lon = parser.getAttributeValue(null, "lon")?.toDoubleOrNull()
                    if (lat != null && lon != null) {
                        waypoints.add(Point.fromLngLat(lon, lat))
                    }
                }
                "trkpt" -> {
                    val lat = parser.getAttributeValue(null, "lat")?.toDoubleOrNull()
                    val lon = parser.getAttributeValue(null, "lon")?.toDoubleOrNull()
                    if (lat != null && lon != null) {
                        trackPoints.add(Point.fromLngLat(lon, lat))
                    }
                }
            }
        }
        eventType = parser.next()
    }

    return GpxData(waypoints, trackPoints)
}

fun toLineString(points: List<Point>): LineString? {
    return if (points.size >= 2) LineString.fromLngLats(points) else null
}

fun latLngBoundsFor(points: List<Point>): LatLngBounds {
    val builder = LatLngBounds.Builder()
    points.forEach { pt ->
        builder.include(LatLng(pt.latitude(), pt.longitude()))
    }
    return builder.build()
}
